<script>

  function updateLessonLink_{{ name|replace('-', '_') }}(selectEl) {

    var btn = document.getElementById("lesson-plan-go-btn-{{ name }}");

    var url = selectEl.value;


    if (url && url !== "#") {

      btn.disabled = false;

      btn.removeAttribute("aria-disabled");

      btn.onclick = function() {

        window.open(url, "_blank", "noopener,noreferrer");

      };

    } else {

      btn.disabled = true;

      btn.setAttribute("aria-disabled", "true");

      btn.onclick = null;

    }

  }

</script>

{% end_require_js %}
